    60→// ============================================================
    61→// Utility: Format date to BM
    62→// ============================================================
    63→function formatDateBM(date: Date | string): string {
    64→  const d = typeof date === 'string' ? new Date(date) : date
    65→  const months = ['Januari', 'Februari', 'Mac', 'April', 'Mei', 'Jun', 'Julai', 'Ogos', 'September', 'Oktober', 'November', 'Disember']
    66→  return `${d.getDate()} ${months[d.getMonth()]} ${d.getFullYear()}`
    67→}
    68→
    69→function formatDateShort(date: Date | string): string {
    70→  const d = typeof date === 'string' ? new Date(date) : date
    71→  return d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' })
    72→}

   154→function PublicPortal() {
   155→  const [mykadInput, setMykadInput] = useState('')
   156→  const [loading, setLoading] = useState(false)
   157→  const [results, setResults] = useState<{ bilangan: number; namaDipaparkan: string; noMykadFormat: string; sijil: SijilItem[] } | null>(null)
   158→  const [error, setError] = useState('')
   159→  const [previewSijil, setPreviewSijil] = useState<SijilItem | null>(null)
   160→  const [pdfUrl, setPdfUrl] = useState<string | null>(null)
   161→  const [pdfLoading, setPdfLoading] = useState(false)
   162→  const { toast } = useToast()
   163→
   164→  // Format MyKad with dashes as user types
   165→  const handleMyKadChange = (value: string) => {
   166→    const digits = value.replace(/[^0-9]/g, '')
   167→    if (digits.length > 12) return
   168→    let formatted = digits
   169→    if (digits.length > 6) formatted = `${digits.slice(0, 6)}-${digits.slice(6)}`
   170→    if (digits.length > 8) formatted = `${digits.slice(0, 6)}-${digits.slice(6, 8)}-${digits.slice(8)}`
   171→    setMykadInput(formatted)
   172→  }
   173→
   174→  const handleSearch = async () => {
   175→    const digits = mykadInput.replace(/[^0-9]/g, '')
   176→    if (digits.length !== 12) {
   177→      setError('Nombor MyKad mesti mengandungi 12 digit.')
   178→      return
   179→    }
   180→    setLoading(true)
   181→    setError('')
   182→    setResults(null)
   183→    try {
   184→      const res = await fetch('/api/awam/carian', {
   185→        method: 'POST',
   186→        headers: { 'Content-Type': 'application/json' },
   187→        body: JSON.stringify({ no_mykad: digits }),
   188→      })
   189→      const data = await res.json()
   190→      if (data.berjaya) {
   191→        setResults(data)
   192→      } else {
   193→        setError(data.mesej || 'Tiada rekod ditemui.')
   194→      }
   195→    } catch {
   196→      setError('Ralat sambungan. Sila cuba sebentar lagi.')
   197→    } finally {
   198→      setLoading(false)
   199→    }
   200→  }
   201→
   202→  const handleDownload = async (sijilId: string) => {
   203→    setPdfLoading(true)
   204→    try {
   205→      const res = await fetch('/api/awam/jana-sijil', {
   206→        method: 'POST',
   207→        headers: { 'Content-Type': 'application/json' },
   208→        body: JSON.stringify({ sijilId }),
   209→      })
   210→      if (!res.ok) throw new Error('Gagal menjana sijil')
   211→      const blob = await res.blob()
   212→      const url = URL.createObjectURL(blob)
   213→      setPdfUrl(url)
   214→      const a = document.createElement('a')
   215→      a.href = url
   216→      a.download = 'Sijil_ADTEC_Sandakan.pdf'
   217→      a.click()
   218→      toast({ title: 'Sijil Dimuat Turun', description: 'Fail PDF sijil telah dimuat turun.' })
   219→    } catch {
   220→      toast({ title: 'Ralat', description: 'Gagal menjana sijil. Sila cuba lagi.', variant: 'destructive' })
   221→    } finally {
   222→      setPdfLoading(false)
   223→    }
   224→  }
   225→
   226→  const handlePreview = async (sijil: SijilItem) => {
   227→    if (!sijil.id) return
   228→    setPreviewSijil(sijil)
   229→    setPdfLoading(true)
   230→    try {
   231→      const res = await fetch('/api/awam/jana-sijil', {
   232→        method: 'POST',
   233→        headers: { 'Content-Type': 'application/json' },
   234→        body: JSON.stringify({ sijilId: sijil.id }),
   235→      })
   236→      if (!res.ok) throw new Error('Gagal')
   237→      const blob = await res.blob()
   238→      const url = URL.createObjectURL(blob)
   239→      setPdfUrl(url)
   240→    } catch {
   241→      toast({ title: 'Ralat', description: 'Gagal pratonton sijil.', variant: 'destructive' })
   242→    } finally {
   243→      setPdfLoading(false)
   244→    }
   245→  }
   246→
   247→  const handlePrint = async (sijilId: string) => {
   248→    setPdfLoading(true)
   249→    try {
   250→      const res = await fetch('/api/awam/jana-sijil', {
   251→        method: 'POST',
   252→        headers: { 'Content-Type': 'application/json' },
   253→        body: JSON.stringify({ sijilId }),
   254→      })
   255→      if (!res.ok) throw new Error('Gagal')
   256→      const blob = await res.blob()
   257→      const url = URL.createObjectURL(blob)
   258→      const printWindow = window.open(url, '_blank')
   259→      if (printWindow) {
   260→        printWindow.onload = () => { printWindow.print() }
   261→      }
   262→    } catch {
   263→      toast({ title: 'Ralat', description: 'Gagal mencetak sijil.', variant: 'destructive' })
   264→    } finally {
   265→      setPdfLoading(false)
   266→    }
   267→  }
   268→
   269→  return (
   270→    <div className="max-w-4xl mx-auto space-y-6">
   271→      {/* Hero Section */}
   272→      <div className="clay-card-lg p-8 sm:p-12 text-center hero-gradient relative overflow-hidden">
   273→        <div className="absolute inset-0 opacity-5" style={{ background: 'radial-gradient(circle at 30% 40%, var(--clay-primary), transparent 50%)' }} />
   274→        <div className="relative z-10">
   275→          <img src="/logo-adtec.png" alt="Logo ADTEC" className="h-16 sm:h-20 mx-auto mb-4 object-contain clay-float" />
   276→          <h1 className="text-2xl sm:text-4xl font-bold mb-2" style={{ color: 'var(--clay-ink)' }}>
   277→            e-Sijil ADTEC Sandakan
   278→          </h1>
   279→          <p className="text-lg sm:text-xl mb-1" style={{ color: 'var(--clay-primary-dark)' }}>
   280→            Sistem Cetak Sijil Digital Dinamik
   281→          </p>
   282→          <p className="text-sm max-w-lg mx-auto" style={{ color: 'var(--clay-ink-secondary)' }}>
   283→            Dapatkan sijil rasmi anda dalam masa kurang 1 minit — hanya dengan memasukkan nombor MyKad.
   284→          </p>
   285→        </div>
   286→      </div>
   287→
   288→      {/* Search Section */}
   289→      <div className="clay-card p-6 sm:p-8">
   290→        <div className="text-center mb-4">
   291→          <h2 className="text-lg font-semibold mb-1" style={{ color: 'var(--clay-ink)' }}>
   292→            Semak Sijil Anda
   293→          </h2>
   294→          <p className="text-sm" style={{ color: 'var(--clay-ink-soft)' }}>
   295→            Masukkan nombor MyKad 12 digit untuk mencari sijil yang layak
   296→          </p>
   297→        </div>
   298→        <div className="flex flex-col sm:flex-row gap-3 max-w-lg mx-auto">
   299→          <div className="flex-1 relative">
   300→            <input
   301→              type="text"
   302→              value={mykadInput}
   303→              onChange={(e) => handleMyKadChange(e.target.value)}
   304→              onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   305→              placeholder="901231-12-5678"
   306→              className="w-full clay-input mykad-input px-6 text-center"
   307→              style={{ background: 'var(--clay)', color: 'var(--clay-ink)', fontSize: '20px', letterSpacing: '2px' }}
   308→              maxLength={14}
   309→            />
   310→            <span className="absolute right-4 top-1/2 -translate-y-1/2 text-xs" style={{ color: 'var(--clay-ink-soft)' }}>
   311→              {mykadInput.replace(/[^0-9]/g, '').length}/12
   312→            </span>
   313→          </div>
   314→          <button
   315→            onClick={handleSearch}
   316→            disabled={loading || mykadInput.replace(/[^0-9]/g, '').length !== 12}
   317→            className="clay-btn flex items-center justify-center gap-2 px-6 disabled:opacity-50 disabled:cursor-not-allowed"
   318→          >
   319→            {loading ? <Loader2 className="w-5 h-5 animate-spin" /> : <Search className="w-5 h-5" />}
   320→            Semak
   321→          </button>
   322→        </div>
   323→        {error && (
   324→          <div className="mt-4 text-center">
   325→            <p className="text-sm px-4 py-2 inline-block rounded-full" style={{ background: 'var(--clay-danger-bg)', color: 'var(--clay-danger)' }}>
   326→              {error}
   327→            </p>
   328→          </div>
   329→        )}
   330→      </div>
   331→
   332→      {/* Results */}
   333→      {results && (
   334→        <div className="space-y-4">
   335→          <div className="flex items-center justify-between px-2">
   336→            <div>
   337→              <h3 className="font-semibold" style={{ color: 'var(--clay-ink)' }}>
   338→                {results.namaDipaparkan}
   339→              </h3>
   340→              <p className="text-xs" style={{ color: 'var(--clay-ink-soft)' }}>
   341→                MyKad: {results.noMykadFormat} · {results.bilangan} sijil ditemui
   342→              </p>
   343→            </div>
   344→            <Badge variant="secondary" className="clay-badge" style={{ color: 'var(--clay-success)' }}>

   421→                        style={{ background: 'var(--clay)', color: 'var(--clay-primary-dark)', borderRadius: '16px', boxShadow: 'var(--clay-shadow-sm)', fontSize: '12px' }}

   434→      {/* Info Section */}

  1149→function TemplateEditor({ template, onSave, onClose }: {
  1150→  template: any
  1151→  onSave: (medan: any[]) => void
  1152→  onClose: () => void
  1153→}) {
  1154→  const [medan, setMedan] = useState<any[]>(template.medanTemplat || [])
  1155→  const [selectedField, setSelectedField] = useState<number | null>(null)
  1156→  const [dragging, setDragging] = useState<number | null>(null)
  1157→  const [dragOffset, setDragOffset] = useState({ x: 0, y: 0 })
  1158→  const canvasRef = useRef<HTMLDivElement>(null)
  1159→  const [saving, setSaving] = useState(false)
  1160→  const { toast } = useToast()
  1161→
  1162→  const isLandscape = template.orientasi === 'landskap'
  1163→
  1164→  // Add a new field
  1165→  const addField = (kunci: string) => {
  1166→    const def = MEDAN_SENARAI.find(m => m.kunci === kunci)
  1167→    if (!def) return
  1168→    const newField = {
  1169→      kunciMedan: kunci,
  1170→      jenisElemen: kunci === 'qr_pengesahan' ? 'qr' : 'teks',
  1171→      posXPeratus: 50,
  1172→      posYPeratus: 50,
  1173→      lebarPeratus: kunci === 'qr_pengesahan' ? 10 : 40,
  1174→      keluargaFon: 'Times New Roman',
  1175→      saizFon: kunci === 'nama_penuh' ? 36 : kunci === 'nama_kursus' ? 24 : 18,
  1176→      warnaFon: '#1a1a2e',
  1177→      gayaFon: kunci === 'nama_penuh' || kunci === 'nama_kursus' ? 'tebal' : 'normal',
  1178→      penjajaran: 'tengah',
  1179→      autoKecil: true,
  1180→    }
  1181→    setMedan([...medan, newField])
  1182→    setSelectedField(medan.length)
  1183→  }
  1184→
  1185→  // Remove a field
  1186→  const removeField = (index: number) => {
  1187→    const newMedan = medan.filter((_: any, i: number) => i !== index)
  1188→    setMedan(newMedan)
  1189→    if (selectedField === index) setSelectedField(null)
  1190→    else if (selectedField !== null && selectedField > index) setSelectedField(selectedField - 1)
  1191→  }
  1192→
  1193→  // Update a field property
  1194→  const updateField = (index: number, key: string, value: any) => {
  1195→    const newMedan = [...medan]
  1196→    newMedan[index] = { ...newMedan[index], [key]: value }
  1197→    setMedan(newMedan)
  1198→  }
  1199→
  1200→  // Mouse handlers for drag
  1201→  const handleMouseDown = (e: React.MouseEvent, index: number) => {
  1202→    e.preventDefault()
  1203→    e.stopPropagation()
  1204→    setDragging(index)
  1205→    setSelectedField(index)
  1206→    const rect = canvasRef.current?.getBoundingClientRect()
  1207→    if (rect) {
  1208→      const fieldEl = e.currentTarget as HTMLElement
  1209→      const fieldRect = fieldEl.getBoundingClientRect()
  1210→      setDragOffset({
  1211→        x: e.clientX - fieldRect.left - fieldRect.width / 2,
  1212→        y: e.clientY - fieldRect.top - fieldRect.height / 2,
  1213→      })
  1214→    }
  1215→  }
  1216→
  1217→  const handleMouseMove = useCallback((e: React.MouseEvent) => {
  1218→    if (dragging === null || !canvasRef.current) return
  1219→    const rect = canvasRef.current.getBoundingClientRect()
  1220→    const x = ((e.clientX - rect.left) / rect.width) * 100
  1221→    const y = ((e.clientY - rect.top) / rect.height) * 100
  1222→    const clampedX = Math.max(5, Math.min(95, x))
  1223→    const clampedY = Math.max(5, Math.min(95, y))
  1224→    setMedan(prev => {
  1225→      const newMedan = [...prev]
  1226→      if (dragging !== null && newMedan[dragging]) {
  1227→        newMedan[dragging] = { ...newMedan[dragging], posXPeratus: Math.round(clampedX * 10) / 10, posYPeratus: Math.round(clampedY * 10) / 10 }
  1228→      }
  1229→      return newMedan
  1230→    })
  1231→  }, [dragging])
  1232→
  1233→  const handleMouseUp = useCallback(() => {
  1234→    setDragging(null)
  1235→  }, [])
  1236→
  1237→  // Save handler
  1238→  const handleSave = async () => {
  1239→    // Validate: must have nama_penuh and nama_kursus
  1240→    const hasNama = medan.some((m: any) => m.kunciMedan === 'nama_penuh')
  1241→    const hasKursus = medan.some((m: any) => m.kunciMedan === 'nama_kursus')
  1242→    if (!hasNama || !hasKursus) {
  1243→      toast({
  1244→        title: 'Medan Wajib Hilang',
  1245→        description: 'Templat mesti mempunyai medan Nama Penuh dan Nama Kursus.',
  1246→        variant: 'destructive',
  1247→      })
  1248→      return
  1249→    }
  1250→    setSaving(true)
  1251→    await onSave(medan)
  1252→    setSaving(false)
  1253→  }
  1254→
  1255→  const selectedFieldData = selectedField !== null ? medan[selectedField] : null
  1256→
  1257→  return (
  1258→    <div className="space-y-4">
  1259→      {/* Header */}
  1260→      <div className="flex items-center justify-between">
  1261→        <div className="flex items-center gap-3">
  1262→          <Button
  1263→            onClick={onClose}
  1264→            className="clay-btn-secondary text-sm px-3 py-2"
  1265→            variant="outline"
  1266→            style={{ background: 'var(--clay)', color: 'var(--clay-primary-dark)', borderRadius: '16px', boxShadow: 'var(--clay-shadow-sm)' }}
  1267→          >
  1268→            <ArrowLeft className="w-4 h-4 mr-1" /> Kembali
  1269→          </Button>
  1270→          <div>
  1271→            <h3 className="font-semibold" style={{ color: 'var(--clay-ink)' }}>Editor Templat Sijil</h3>
  1272→            <p className="text-xs" style={{ color: 'var(--clay-ink-soft)' }}>{template.namaTemplat}</p>
  1273→          </div>
  1274→        </div>
  1275→        <Button
  1276→          onClick={handleSave}
  1277→          disabled={saving}
  1278→          className="clay-btn text-sm px-5 py-2 flex items-center gap-2"
  1279→          style={{ background: 'var(--clay-primary)', color: 'white', borderRadius: '20px', boxShadow: 'var(--clay-shadow-sm)' }}
  1280→        >
  1281→          {saving ? <Loader2 className="w-4 h-4 animate-spin" /> : <Save className="w-4 h-4" />}
  1282→          {saving ? 'Menyimpan...' : 'Simpan Templat'}
  1283→        </Button>
  1284→      </div>
  1285→
  1286→      {/* Main Editor Area */}
  1287→      <div className="flex flex-col lg:flex-row gap-4">
  1288→        {/* Canvas */}
  1289→        <div className="flex-1">
  1290→          <div className="clay-card-lg p-4">
  1291→            <div className="flex items-center justify-between mb-3">
  1292→              <span className="text-xs font-medium" style={{ color: 'var(--clay-ink-soft)' }}>
  1293→                Kanvas Templat — {isLandscape ? 'Landskap' : 'Potret'} {template.saizKertas?.toUpperCase()}
  1294→              </span>
  1295→              <span className="text-xs" style={{ color: 'var(--clay-ink-soft)' }}>
  1296→                Seret medan untuk mengubah kedudukan
  1297→              </span>
  1298→            </div>
  1299→            {/* Canvas */}
  1300→            <div
  1301→              ref={canvasRef}
  1302→              className="relative mx-auto rounded-lg overflow-hidden cursor-crosshair"
  1303→              style={{
  1304→                width: '100%',
  1305→                aspectRatio: isLandscape ? '297/210' : '210/297',
  1306→                background: 'linear-gradient(135deg, #fafbff 0%, #f0f1f8 50%, #e8eaf4 100%)',
  1307→                border: '2px solid var(--border)',
  1308→                boxShadow: 'inset 0 2px 8px rgba(120,124,170,0.1)',
  1309→              }}
  1310→              onMouseMove={handleMouseMove}
  1311→              onMouseUp={handleMouseUp}
  1312→              onMouseLeave={handleMouseUp}
