    62→// ============================================================
    63→// Utility: Format date to BM
    64→// ============================================================
    65→function formatDateBM(date: Date | string): string {
    66→  const d = typeof date === 'string' ? new Date(date) : date
    67→  const months = ['Januari', 'Februari', 'Mac', 'April', 'Mei', 'Jun', 'Julai', 'Ogos', 'September', 'Oktober', 'November', 'Disember']
    68→  return `${d.getDate()} ${months[d.getMonth()]} ${d.getFullYear()}`
    69→}
    70→
    71→function formatDateShort(date: Date | string): string {
    72→  const d = typeof date === 'string' ? new Date(date) : date
    73→  return d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' })
    74→}

   803→function AdminLogin({ onLogin }: { onLogin: (user: AdminUser) => void }) {
   804→  const [emel, setEmel] = useState('')
   805→  const [kataLaluan, setKataLaluan] = useState('')
   806→  const [loading, setLoading] = useState(false)
   807→  const [error, setError] = useState('')
   808→  const { toast } = useToast()
   809→
   810→  const handleLogin = async () => {
   811→    setLoading(true)
   812→    setError('')
   813→    try {
   814→      const res = await fetch('/api/auth/login', {
   815→        method: 'POST',
   816→        headers: { 'Content-Type': 'application/json' },
   817→        body: JSON.stringify({ emel, kataLaluan }),
   818→      })
   819→      const data = await res.json()
   820→      if (data.berjaya) {
   821→        onLogin(data.pengguna)
   822→        toast({ title: 'Log Masuk Berjaya', description: `Selamat datang, ${data.pengguna.namaPenuh}` })
   823→      } else {
   824→        setError(data.mesej)
   825→      }
   826→    } catch {
   827→      setError('Ralat sambungan.')
   828→    } finally {
   829→      setLoading(false)
   830→    }
   831→  }
   832→
   833→  return (
   834→    <div className="max-w-md mx-auto">
   835→      <div className="clay-card-lg p-8">
   836→        <div className="text-center mb-6">
   837→          <img src="/logo-adtec.png" alt="ADTEC" className="h-16 mx-auto mb-4" />
   838→          <h2 className="text-xl font-bold" style={{ color: 'var(--clay-ink)' }}>Log Masuk Pentadbir</h2>
   839→          <p className="text-sm" style={{ color: 'var(--clay-ink-soft)' }}>Portal Pengurusan e-Sijil</p>
   840→        </div>
   841→        <div className="space-y-4">
   842→          <div>
   843→            <label className="text-sm font-medium mb-1 block" style={{ color: 'var(--clay-ink-secondary)' }}>E-mel Rasmi</label>
   844→            <input
   845→              type="email"
   846→              value={emel}
   847→              onChange={(e) => setEmel(e.target.value)}
   848→              onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
   849→              placeholder="admin@adtec.gov.my"
   850→              className="w-full clay-input px-4 py-3"
   851→              style={{ background: 'var(--clay)', color: 'var(--clay-ink)' }}
   852→            />
   853→          </div>
   854→          <div>
   855→            <label className="text-sm font-medium mb-1 block" style={{ color: 'var(--clay-ink-secondary)' }}>Kata Laluan</label>
   856→            <input
   857→              type="password"
   858→              value={kataLaluan}
   859→              onChange={(e) => setKataLaluan(e.target.value)}
   860→              onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
   861→              placeholder="••••••••••••"
   862→              className="w-full clay-input px-4 py-3"
   863→              style={{ background: 'var(--clay)', color: 'var(--clay-ink)' }}
   864→            />
   865→          </div>
   866→          {error && <p className="text-sm text-center" style={{ color: 'var(--clay-danger)' }}>{error}</p>}
   867→          <button
   868→            onClick={handleLogin}
