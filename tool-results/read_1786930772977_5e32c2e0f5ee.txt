    63→// ============================================================
    64→// Utility: Format date to BM
    65→// ============================================================
    66→function formatDateBM(date: Date | string): string {
    67→  const d = typeof date === 'string' ? new Date(date) : date
    68→  const months = ['Januari', 'Februari', 'Mac', 'April', 'Mei', 'Jun', 'Julai', 'Ogos', 'September', 'Oktober', 'November', 'Disember']
    69→  return `${d.getDate()} ${months[d.getMonth()]} ${d.getFullYear()}`
    70→}
    71→
    72→function formatDateShort(date: Date | string): string {
    73→  const d = typeof date === 'string' ? new Date(date) : date
    74→  return d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' })
    75→}

   804→function AdminLogin({ onLogin }: { onLogin: (user: AdminUser) => void }) {
   805→  const [emel, setEmel] = useState('')
   806→  const [kataLaluan, setKataLaluan] = useState('')
   807→  const [loading, setLoading] = useState(false)
   808→  const [error, setError] = useState('')
   809→  const { toast } = useToast()
   810→
   811→  const handleLogin = async () => {
   812→    setLoading(true)
   813→    setError('')
   814→    try {
   815→      const res = await fetch('/api/auth/login', {
   816→        method: 'POST',
   817→        headers: { 'Content-Type': 'application/json' },
   818→        body: JSON.stringify({ emel, kataLaluan }),
   819→      })
   820→      const data = await res.json()
   821→      if (data.berjaya) {
   822→        onLogin(data.pengguna)
   823→        toast({ title: 'Log Masuk Berjaya', description: `Selamat datang, ${data.pengguna.namaPenuh}` })
   824→      } else {
   825→        setError(data.mesej)
   826→      }
   827→    } catch {
   828→      setError('Ralat sambungan.')
   829→    } finally {
   830→      setLoading(false)
   831→    }
   832→  }
   833→
   834→  return (
   835→    <div className="max-w-md mx-auto">
   836→      <div className="clay-card-lg p-8">
   837→        <div className="text-center mb-6">
   838→          <img src="/logo-adtec.png" alt="ADTEC" className="h-16 mx-auto mb-4" />
   839→          <h2 className="text-xl font-bold" style={{ color: 'var(--clay-ink)' }}>Log Masuk Pentadbir</h2>
   840→          <p className="text-sm" style={{ color: 'var(--clay-ink-soft)' }}>Portal Pengurusan e-Sijil</p>
   841→        </div>
   842→        <div className="space-y-4">
   843→          <div>
   844→            <label className="text-sm font-medium mb-1 block" style={{ color: 'var(--clay-ink-secondary)' }}>E-mel Rasmi</label>
   845→            <input
   846→              type="email"
   847→              value={emel}
   848→              onChange={(e) => setEmel(e.target.value)}
   849→              onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
   850→              placeholder="admin@adtec.gov.my"
   851→              className="w-full clay-input px-4 py-3"
   852→              style={{ background: 'var(--clay)', color: 'var(--clay-ink)' }}
   853→            />
   854→          </div>
   855→          <div>
   856→            <label className="text-sm font-medium mb-1 block" style={{ color: 'var(--clay-ink-secondary)' }}>Kata Laluan</label>
   857→            <input
   858→              type="password"
   859→              value={kataLaluan}
   860→              onChange={(e) => setKataLaluan(e.target.value)}
   861→              onKeyDown={(e) => e.key === 'Enter' && handleLogin()}
   862→              placeholder="••••••••••••"
   863→              className="w-full clay-input px-4 py-3"
   864→              style={{ background: 'var(--clay)', color: 'var(--clay-ink)' }}
   865→            />
   866→          </div>
   867→          {error && <p className="text-sm text-center" style={{ color: 'var(--clay-danger)' }}>{error}</p>}
   868→          <button
