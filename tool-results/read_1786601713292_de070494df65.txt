    57→// ============================================================
    58→// Utility: Format date to BM
    59→// ============================================================
    60→function formatDateBM(date: Date | string): string {
    61→  const d = typeof date === 'string' ? new Date(date) : date
    62→  const months = ['Januari', 'Februari', 'Mac', 'April', 'Mei', 'Jun', 'Julai', 'Ogos', 'September', 'Oktober', 'November', 'Disember']
    63→  return `${d.getDate()} ${months[d.getMonth()]} ${d.getFullYear()}`
    64→}
    65→
    66→function formatDateShort(date: Date | string): string {
    67→  const d = typeof date === 'string' ? new Date(date) : date
    68→  return d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' })
    69→}

   151→function PublicPortal() {
   152→  const [mykadInput, setMykadInput] = useState('')
   153→  const [loading, setLoading] = useState(false)
   154→  const [results, setResults] = useState<{ bilangan: number; namaDipaparkan: string; noMykadFormat: string; sijil: SijilItem[] } | null>(null)
   155→  const [error, setError] = useState('')
   156→  const [previewSijil, setPreviewSijil] = useState<SijilItem | null>(null)
   157→  const [pdfUrl, setPdfUrl] = useState<string | null>(null)
   158→  const [pdfLoading, setPdfLoading] = useState(false)
   159→  const { toast } = useToast()
   160→
   161→  // Format MyKad with dashes as user types
   162→  const handleMyKadChange = (value: string) => {
   163→    const digits = value.replace(/[^0-9]/g, '')
   164→    if (digits.length > 12) return
   165→    let formatted = digits
   166→    if (digits.length > 6) formatted = `${digits.slice(0, 6)}-${digits.slice(6)}`
   167→    if (digits.length > 8) formatted = `${digits.slice(0, 6)}-${digits.slice(6, 8)}-${digits.slice(8)}`
   168→    setMykadInput(formatted)
   169→  }
   170→
   171→  const handleSearch = async () => {
   172→    const digits = mykadInput.replace(/[^0-9]/g, '')
   173→    if (digits.length !== 12) {
   174→      setError('Nombor MyKad mesti mengandungi 12 digit.')
   175→      return
   176→    }
   177→    setLoading(true)
   178→    setError('')
   179→    setResults(null)
   180→    try {
   181→      const res = await fetch('/api/awam/carian', {
   182→        method: 'POST',
   183→        headers: { 'Content-Type': 'application/json' },
   184→        body: JSON.stringify({ no_mykad: digits }),
   185→      })
   186→      const data = await res.json()
   187→      if (data.berjaya) {
   188→        setResults(data)
   189→      } else {
   190→        setError(data.mesej || 'Tiada rekod ditemui.')
   191→      }
   192→    } catch {
   193→      setError('Ralat sambungan. Sila cuba sebentar lagi.')
   194→    } finally {
   195→      setLoading(false)
   196→    }
   197→  }
   198→
   199→  const handleDownload = async (sijilId: string) => {
   200→    setPdfLoading(true)
   201→    try {
   202→      const res = await fetch('/api/awam/jana-sijil', {
   203→        method: 'POST',
   204→        headers: { 'Content-Type': 'application/json' },
   205→        body: JSON.stringify({ sijilId }),
   206→      })
   207→      if (!res.ok) throw new Error('Gagal menjana sijil')
   208→      const blob = await res.blob()
   209→      const url = URL.createObjectURL(blob)
   210→      setPdfUrl(url)
   211→      const a = document.createElement('a')
   212→      a.href = url
   213→      a.download = 'Sijil_ADTEC_Sandakan.pdf'
   214→      a.click()
   215→      toast({ title: 'Sijil Dimuat Turun', description: 'Fail PDF sijil telah dimuat turun.' })
   216→    } catch {
   217→      toast({ title: 'Ralat', description: 'Gagal menjana sijil. Sila cuba lagi.', variant: 'destructive' })
   218→    } finally {
   219→      setPdfLoading(false)
   220→    }
   221→  }
   222→
   223→  const handlePreview = async (sijil: SijilItem) => {
   224→    if (!sijil.id) return
   225→    setPreviewSijil(sijil)
   226→    setPdfLoading(true)
   227→    try {
   228→      const res = await fetch('/api/awam/jana-sijil', {
   229→        method: 'POST',
   230→        headers: { 'Content-Type': 'application/json' },
   231→        body: JSON.stringify({ sijilId: sijil.id }),
   232→      })
   233→      if (!res.ok) throw new Error('Gagal')
   234→      const blob = await res.blob()
   235→      const url = URL.createObjectURL(blob)
   236→      setPdfUrl(url)
   237→    } catch {
   238→      toast({ title: 'Ralat', description: 'Gagal pratonton sijil.', variant: 'destructive' })
   239→    } finally {
   240→      setPdfLoading(false)
   241→    }
   242→  }
   243→
   244→  const handlePrint = async (sijilId: string) => {
   245→    setPdfLoading(true)
   246→    try {
   247→      const res = await fetch('/api/awam/jana-sijil', {
   248→        method: 'POST',
   249→        headers: { 'Content-Type': 'application/json' },
   250→        body: JSON.stringify({ sijilId }),
   251→      })
   252→      if (!res.ok) throw new Error('Gagal')
   253→      const blob = await res.blob()
   254→      const url = URL.createObjectURL(blob)
   255→      const printWindow = window.open(url, '_blank')
   256→      if (printWindow) {
   257→        printWindow.onload = () => { printWindow.print() }
   258→      }
   259→    } catch {
   260→      toast({ title: 'Ralat', description: 'Gagal mencetak sijil.', variant: 'destructive' })
   261→    } finally {
   262→      setPdfLoading(false)
   263→    }
   264→  }
   265→
   266→  return (
   267→    <div className="max-w-4xl mx-auto space-y-6">
   268→      {/* Hero Section */}
   269→      <div className="clay-card-lg p-8 sm:p-12 text-center hero-gradient relative overflow-hidden">
   270→        <div className="absolute inset-0 opacity-5" style={{ background: 'radial-gradient(circle at 30% 40%, var(--clay-primary), transparent 50%)' }} />
   271→        <div className="relative z-10">
   272→          <img src="/logo-adtec.png" alt="Logo ADTEC" className="h-16 sm:h-20 mx-auto mb-4 object-contain clay-float" />
   273→          <h1 className="text-2xl sm:text-4xl font-bold mb-2" style={{ color: 'var(--clay-ink)' }}>
   274→            e-Sijil ADTEC Sandakan
   275→          </h1>
   276→          <p className="text-lg sm:text-xl mb-1" style={{ color: 'var(--clay-primary-dark)' }}>
   277→            Sistem Cetak Sijil Digital Dinamik
   278→          </p>
   279→          <p className="text-sm max-w-lg mx-auto" style={{ color: 'var(--clay-ink-secondary)' }}>
   280→            Dapatkan sijil rasmi anda dalam masa kurang 1 minit — hanya dengan memasukkan nombor MyKad.
   281→          </p>
   282→        </div>
   283→      </div>
   284→
   285→      {/* Search Section */}
   286→      <div className="clay-card p-6 sm:p-8">
   287→        <div className="text-center mb-4">
   288→          <h2 className="text-lg font-semibold mb-1" style={{ color: 'var(--clay-ink)' }}>
   289→            Semak Sijil Anda
   290→          </h2>
   291→          <p className="text-sm" style={{ color: 'var(--clay-ink-soft)' }}>
   292→            Masukkan nombor MyKad 12 digit untuk mencari sijil yang layak
   293→          </p>
   294→        </div>
   295→        <div className="flex flex-col sm:flex-row gap-3 max-w-lg mx-auto">
   296→          <div className="flex-1 relative">
   297→            <input
   298→              type="text"
   299→              value={mykadInput}
   300→              onChange={(e) => handleMyKadChange(e.target.value)}
   301→              onKeyDown={(e) => e.key === 'Enter' && handleSearch()}
   302→              placeholder="901231-12-5678"
   303→              className="w-full clay-input mykad-input px-6 text-center"
   304→              style={{ background: 'var(--clay)', color: 'var(--clay-ink)', fontSize: '20px', letterSpacing: '2px' }}
   305→              maxLength={14}
   306→            />
   307→            <span className="absolute right-4 top-1/2 -translate-y-1/2 text-xs" style={{ color: 'var(--clay-ink-soft)' }}>
   308→              {mykadInput.replace(/[^0-9]/g, '').length}/12
   309→            </span>
   310→          </div>
   311→          <button
   312→            onClick={handleSearch}
   313→            disabled={loading || mykadInput.replace(/[^0-9]/g, '').length !== 12}
   314→            className="clay-btn flex items-center justify-center gap-2 px-6 disabled:opacity-50 disabled:cursor-not-allowed"
   315→          >
   316→            {loading ? <Loader2 className="w-5 h-5 animate-spin" /> : <Search className="w-5 h-5" />}
   317→            Semak
   318→          </button>
   319→        </div>
   320→        {error && (
   321→          <div className="mt-4 text-center">
   322→            <p className="text-sm px-4 py-2 inline-block rounded-full" style={{ background: 'var(--clay-danger-bg)', color: 'var(--clay-danger)' }}>
   323→              {error}
   324→            </p>
   325→          </div>
   326→        )}
   327→      </div>
   328→
   329→      {/* Results */}
   330→      {results && (
   331→        <div className="space-y-4">
   332→          <div className="flex items-center justify-between px-2">
   333→            <div>
   334→              <h3 className="font-semibold" style={{ color: 'var(--clay-ink)' }}>
   335→                {results.namaDipaparkan}
   336→              </h3>
   337→              <p className="text-xs" style={{ color: 'var(--clay-ink-soft)' }}>
   338→                MyKad: {results.noMykadFormat} · {results.bilangan} sijil ditemui

   341→            <Badge variant="secondary" className="clay-badge" style={{ color: 'var(--clay-success)' }}>

   354→                    <h4 className="font-semibold text-base mb-1" style={{ color: 'var(--clay-ink)' }}>

   418→                        style={{ background: 'var(--clay)', color: 'var(--clay-primary-dark)', borderRadius: '16px', boxShadow: 'var(--clay-shadow-sm)', fontSize: '12px' }}

   430→
   431→      {/* Info Section */}

   495→function AdminLogin({ onLogin }: { onLogin: (user: AdminUser) => void }) {
   496→  const [emel, setEmel] = useState('')
   497→  const [kataLaluan, setKataLaluan] = useState('')

   501→
   502→  const handleLogin = async () => {
   503→    setLoading(true)
   504→    setError('')

   506→      const res = await fetch('/api/auth/login', {
   507→        method: 'POST',
   508→        headers: { 'Content-Type': 'application/json' },
   509→        body: JSON.stringify({ emel, kataLaluan }),
